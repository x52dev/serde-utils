1//! String trimming adapters for `deser`.
2//!
3//! Use `#[deser(deserialize_as = Trim)]` for strings, `Option<Trim>` for
4//! optional strings, and `Vec<Trim>` or `HashSet<Trim>` for collections.
5//! For a borrowed `Cow<str>`, use `deser::adapters::TrimWhitespace<deser::adapters::Borrowed>`.
6//!
7//! ```
8//! #[derive(Debug, deser::Deserialize)]
9//! struct Form {
10//!     #[deser(deserialize_as = detrim::deser::NonEmpty)]
11//!     name: String,
12//! }
13//!
14//! let form = deser_json::from_str::<Form>(r#"{"name":"  ferris  "}"#).unwrap();
15//! assert_eq!(form.name, "ferris");
16//! ```
17
18use alloc::{borrow::ToOwned as _, string::String};

20/// Trims whitespace from strings, including string slices.
21///
22/// Compose with `Option`, `Vec`, or `HashSet` to trim their string values.
23pub use ::deser::adapters::TrimWhitespace as Trim;
24use ::deser::{
25    adapters::DeserializeAs,
26    de::{Sink, SinkHandle},
27    Atom, Error, ErrorKind, State,
28};

30/// Trims a string and returns an error if it is empty.
31pub struct NonEmpty;
32
33/// Trims an optional string and converts empty strings to `None`.
34pub struct EmptyAsNone;
35
36::deser::make_slot_wrapper!(TrimSlot);
37
38impl<'de> Sink<'de> for TrimSlot<String> {
39    fn atom(&mut self, atom: Atom<'_>, state: &mut State) -> Result<(), Error> {
40        match atom {
41            Atom::Str(value) if value.trim().is_empty() => Err(Error::new(
42                ErrorKind::Unexpected,
43                "expected a non-empty string",
44            )),
45
46            Atom::Str(value) => {
47                **self = Some(value.trim().to_owned());
48                Ok(())
49            }
50
51            other => self.unexpected_atom(other, state),
52        }
53    }
54}
55
56impl<'de> DeserializeAs<'de, String> for NonEmpty {
57    fn deserialize_into_as<'out>(
58        out: &'out mut Option<String>,
59        _state: &mut State,
60    ) -> SinkHandle<'out, 'de> {
61        TrimSlot::make_handle(out)
62    }
63}
64
65impl<'de> Sink<'de> for TrimSlot<Option<String>> {
66    fn atom(&mut self, atom: Atom<'_>, state: &mut State) -> Result<(), Error> {
67        match atom {
68            Atom::Null => {
69                **self = Some(None);
70                Ok(())
71            }
72
73            Atom::Str(value) => {
74                **self = Some((!value.trim().is_empty()).then(|| value.trim().to_owned()));
75                Ok(())
76            }
77
78            other => self.unexpected_atom(other, state),
79        }
80    }
81}
82
83impl<'de> DeserializeAs<'de, Option<String>> for EmptyAsNone {
84    fn deserialize_into_as<'out>(
85        out: &'out mut Option<Option<String>>,
86        _state: &mut State,
87    ) -> SinkHandle<'out, 'de> {
88        TrimSlot::make_handle(out)
89    }
90
91    fn initial_value_as() -> Option<Option<String>> {
92        Some(None)
93    }
