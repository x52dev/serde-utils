11impl<'de> Sink<'de> for DoubleIntSlot<DoubleInt> {
12    fn atom(&mut self, atom: Atom<'_>, state: &mut State) -> Result<(), Error> {
13        let value = match atom {
14            Atom::I64(value) => i128::from(value),
15            Atom::U64(value) => i128::from(value),
16            other => return self.unexpected_atom(other, state),
17        };
18
19        if !(DoubleInt::MIN..=DoubleInt::MAX).contains(&value) {
20            return Err(Error::new(
21                ErrorKind::Unexpected,
22                "expected an integer between -9007199254740991 and 9007199254740991",
23            ));
24        }
25
26        **self = Some(DoubleInt(value as i64));
27        Ok(())
28    }
29}
