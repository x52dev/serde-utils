1//! Decimal wrappers for miniserde models.
2//!
3//! Use [`Str`] to preserve decimal precision in a JSON string, or [`Float`] for a
4//! JSON number. Both accept strings and numbers during deserialization.
5//! Numeric input and [`Float`] output can lose precision because miniserde uses
6//! `f64` for non-integer JSON numbers. Arbitrary-precision JSON numbers are not supported.
7//!
8//! The nullable, non-required, and double-option wrappers keep the same
9//! deserialization rules as the Serde adapters. Miniserde's derived serializer
10//! writes absent optional values as null. To omit a field, implement
11//! `miniserde::ser::Map` and skip the field when its outer option is `None`.
12//!
13//! ```
14//! use serde_decimal::miniserde::{DoubleOptionStr, NullableFloat};
15//!
16//! #[derive(miniserde::Deserialize, miniserde::Serialize)]
17//! struct Prices {
18//!     price: NullableFloat,
19//!     discount: DoubleOptionStr,
20//! }
21//!
22//! let prices: Prices = miniserde::json::from_str(r#"{"price":null}"#).unwrap();
23//! assert_eq!(prices.price.0, None);
24//! assert_eq!(prices.discount.0, None);
25//! ```
26

37/// Serializes a required decimal as a JSON string.
38#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
39pub struct Str(pub Decimal);
40
41/// Serializes a required decimal as a JSON number, with possible precision loss.
42#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
43pub struct Float(pub Decimal);
44
45/// A required field that accepts null and serializes decimals as JSON strings.
46#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
47pub struct NullableStr(pub Option<Decimal>);
48
49/// A required field that accepts null and serializes decimals as JSON numbers.
50#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
51pub struct NullableFloat(pub Option<Decimal>);
52
53/// A field that can be missing, rejects null, and serializes decimals as JSON strings.
54#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
55pub struct NonRequiredStr(pub Option<Decimal>);
56
57/// A field that can be missing, rejects null, and serializes decimals as JSON numbers.
58#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
59pub struct NonRequiredFloat(pub Option<Decimal>);
60
61/// A string-form decimal field that distinguishes missing, null, and a value.
62///
63/// Missing fields produce `None`; null produces `Some(None)`.
64#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
65pub struct DoubleOptionStr(pub Option<Option<Decimal>>);
66
67/// A number-form decimal field that distinguishes missing, null, and a value.
68///
69/// Missing fields produce `None`; null produces `Some(None)`.
70#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
71pub struct DoubleOptionFloat(pub Option<Option<Decimal>>);
72
73fn serialize_str(value: Decimal) -> Fragment<'static> {
74    Fragment::Str(Cow::Owned(value.to_string()))
75}
76
77fn serialize_float(value: Decimal) -> Fragment<'static> {
78    // Every Decimal is finite and within the range of f64.
79    Fragment::F64(value.to_f64().expect("Decimal fits in f64"))
80}

82macro_rules! decimal_impls {
83    ($ty:ident, $wrap:expr, $unwrap:expr, $serialize:ident $(, $default:expr)? $(; $null:expr)?) => {
84        impl Deserialize for $ty {
85            fn begin(out: &mut Option<Self>) -> &mut dyn Visitor {
86                Place::new(out)
87            }
88
89            $(
90                fn default() -> Option<Self> {
91                    Some($default)
92                }
93            )?
94        }
95
96        impl Visitor for Place<$ty> {
97            $(
98                fn null(&mut self) -> Result<()> {
99                    self.out = Some($null);
100
101                    Ok(())
102                }
103            )?
104
105            fn string(&mut self, value: &str) -> Result<()> {
106                let value = Decimal::from_str(value)
107                    .or_else(|_| Decimal::from_scientific(value))
108                    .map_err(|_| Error)?;
109                self.out = Some(($wrap)(value));
110
111                Ok(())
112            }
113
114            fn negative(&mut self, value: i64) -> Result<()> {
115                self.out = Some(($wrap)(Decimal::from(value)));
116
117                Ok(())
118            }
119
120            fn nonnegative(&mut self, value: u64) -> Result<()> {
121                self.out = Some(($wrap)(Decimal::from(value)));
122
123                Ok(())
124            }
125
126            fn float(&mut self, value: f64) -> Result<()> {
127                let value = Decimal::from_f64(value).ok_or(Error)?;
128                self.out = Some(($wrap)(value));
129
130                Ok(())
131            }
132        }
133
134        impl Serialize for $ty {
135            fn begin(&self) -> Fragment<'_> {
136                match ($unwrap)(self) {
137                    Some(value) => $serialize(value),
138                    None => Fragment::Null,
139                }
140            }
141        }
142    };
143}

145decimal_impls!(Str, Str, |value: &Str| Some(value.0), serialize_str);
146decimal_impls!(Float, Float, |value: &Float| Some(value.0), serialize_float);
